cmake_minimum_required(VERSION 3.22.1)
project(substrate)
set(NODE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../vendor/node")
add_library(node SHARED IMPORTED)
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_ROOT}/bin/${ANDROID_ABI}/libnode.so")
add_library(substrate SHARED node_bridge.cpp)
target_include_directories(substrate PRIVATE "${NODE_ROOT}/include/node")
target_link_libraries(substrate node log)
target_link_options(substrate PRIVATE "-Wl,-z,max-page-size=16384")
